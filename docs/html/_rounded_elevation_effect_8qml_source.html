4import QtQuick.Controls.Material
5import QtQuick.Controls.Material.impl
6
7ElevationEffect {
8 required property int roundedScale
9
10 _shadows: roundedScale === Material.NotRounded ? _defaultShadows : roundedShadows()
11
12 function roundedShadows() {
13 // Make a deep copy.
14 let shadows = [..._defaultShadows]
15 for (let i = 0, strength = 0.95; i < shadows.length; ++i) {
16 // See comment on BoxShadow's strength property for why we do this.
17 shadows[i].strength = strength
18 // We don't want the strength to be too high for the controls with very slightly rounded
19 // corners, as they are quite close to the non-rounded ones in terms of not needing adjustments.
20 // This is still not great for the higher elevations for ExtraSmallScale, but it's as good
21 // as I can get it.
22 strength = Math.max(0.05, strength - (roundedScale > Material.ExtraSmallScale ? 0.1 : 0.3))
23
24 // The values at index 0 are already 0, and we don't want our Math.max(1, ...) code to affect them.
25 if (i > 0) {
26 // The blur values for e.g. buttons with rounded corners are too large, so we reduce them.
27 for (let angularShadowIndex = 0; angularShadowIndex < shadows[i].angularValues.length; ++angularShadowIndex) {
28 shadows[i].angularValues[angularShadowIndex].blur =
29 Math.max(1, Math.floor(shadows[i].angularValues[angularShadowIndex].blur / 4))
30 }
31 }
32 }
33 return shadows
34 }
35}
